// Shoulder Surgery Fragment Shader
// Variant of mtFragmentShader.txt adapted for shoulder anatomy
// Supports materials: skin(2), dermis(3), fat(4), fascia(5),
//   muscle(6), periosteum(7-8), tendon(11), capsule(12), bone(13)

vec3 mod289(vec3 x) {
  return x - floor(x * (1.0 / 289.0)) * 289.0;}
vec2 mod289(vec2 x) {
  return x - floor(x * (1.0 / 289.0)) * 289.0;}
vec3 permute(vec3 x) {
  return mod289(((x*34.0)+1.0)*x);}
vec4 taylorInvSqrt(vec4 r){
		return 1.79284291400159 - 0.85373472095314 * r;}
float snoise(vec2 v)  {
	const vec4 C = vec4(0.211324865405187,  // (3.0-sqrt(3.0))/6.0
		0.366025403784439,  // 0.5*(sqrt(3.0)-1.0)
		-0.577350269189626,  // -1.0 + 2.0 * C.x
		0.024390243902439); // 1.0 / 41.0
	// First corner
	vec2 i  = floor(v + dot(v, C.yy) );
	vec2 x0 = v -   i + dot(i, C.xx);
	// Other corners
	vec2 i1;
	//i1.x = step( x0.y, x0.x ); // x0.x > x0.y ? 1.0 : 0.0
	//i1.y = 1.0 - i1.x;
	i1 = (x0.x > x0.y) ? vec2(1.0, 0.0) : vec2(0.0, 1.0);
	// x0 = x0 - 0.0 + 0.0 * C.xx ;
	// x1 = x0 - i1 + 1.0 * C.xx ;
	// x2 = x0 - 1.0 + 2.0 * C.xx ;
	vec4 x12 = x0.xyxy + C.xxzz;
	x12.xy -= i1;
	// Permutations
	i = mod289(i); // Avoid truncation effects in permutation
	vec3 p = permute( permute( i.y + vec3(0.0, i1.y, 1.0 ))
		+ i.x + vec3(0.0, i1.x, 1.0 ));
	vec3 m = max(0.5 - vec3(dot(x0,x0), dot(x12.xy,x12.xy), dot(x12.zw,x12.zw)), 0.0);
	m = m*m ;
	m = m*m ;
	// Gradients: 41 points uniformly over a line, mapped onto a diamond.
	// The ring size 17*17 = 289 is close to a multiple of 41 (41*7 = 287)
	vec3 x = 2.0 * fract(p * C.www) - 1.0;
	vec3 h = abs(x) - 0.5;
	vec3 ox = floor(x + 0.5);
	vec3 a0 = x - ox;
	// Normalise gradients implicitly by scaling m
	// Approximation of: m *= inversesqrt( a0*a0 + h*h );
	m *= 1.79284291400159 - 0.85373472095314 * ( a0*a0 + h*h );
	// Compute final noise value at P
	vec3 g;
	g.x  = a0.x  * x0.x  + h.x  * x0.y;
	g.yz = a0.yz * x12.xz + h.yz * x12.yw;
	return 130.0 * dot(m, g);}

// next routine gets fat from 4 surrounding noise values [-1,1] - Court Cutting
void getFat(in vec4 nei, out vec4 fatColor, out vec3 normDelta, out float specMult) {
	float h;
	for(int i=0; i<4; ++i){
		h = 1.0 - abs(nei[i]);
		h *= h;
		nei[i] = 1.0 - h; }
	h = 0;
	for(int i=0; i<4; ++i)
		h += nei[i];
	h *= 0.25;
	vec2 p;
	p.x = nei[1]-nei[0];
	p.y = nei[2]-nei[3];
	p *= 130.0;
	p = clamp(p,-1.0,1.0);
	float d,f;
	d = dot(p,p);
	f = inversesqrt(d+1.0);
	p.x = -p.x;
	normDelta = vec3(p,1.0)*f;
	float fatRed, fatGreen, fatBlue;
	if(h<0.04) {
		specMult = 0.2;
		fatRed = (1.0-h)*0.4;
		fatBlue = 0.0;
		fatGreen = (1.0-h)*0.2; }
	else {
		specMult = 1.0;
		fatRed = 0.5 + h*0.8;
		fatBlue = 0.15 + h*0.3;
		fatGreen = 0.35 + h*0.8; }
	fatColor = vec4(fatRed, fatGreen, fatBlue, 1.0); }

out vec4 vFragColor;
uniform vec4 ambientColor;
uniform vec4 diffuseColor;
uniform sampler2D colorMap;
uniform sampler2D normalMap;
uniform sampler2D texture2;
uniform sampler2D texture3;
uniform int material;
uniform mat3   normalMatrix;
smooth in vec3 vLightDir;
smooth in vec3 vEyeDir;
smooth in vec2 vTexCoords;
void main(void)
{
// always lighting with white light
	const float ambientVal = 0.1;
	const float	fatIncr = 0.5/1024.0;
	vec2 fatD = vec2(fatIncr,0.0);
	vec2 faceUV;
	float lightVal = ambientVal;
	float sn,h,dm=6.0,specMult = 0.5;
	const float diffuseVal = 0.9;
	vec3 normDelta = vec3(0.0, 0.0, 1.0);
	vec3 litColor = vec3(1.0, 1.0, 1.0);
	vec4 nei;
	if(material>12) {  // bone surface (material 13)
		// Ivory/cream with subtle porous surface variation
		faceUV = vTexCoords * 200.0;
		sn = snoise(faceUV);
		float pore = snoise(faceUV * 3.5);
		// Subtle porous variation on an ivory base
		float boneBase = 0.92 + 0.03 * sn;
		float porosity = 0.02 * pore;
		vFragColor = vec4(
			boneBase + porosity,
			boneBase - 0.04 + porosity,
			boneBase - 0.10 + porosity * 0.5,
			1.0);
		specMult = 0.4;
	}
	else if(material>11) {  // joint capsule (material 12)
		// Translucent pinkish-white, smooth surface
		faceUV = vTexCoords * 120.0;
		sn = snoise(faceUV);
		float smooth_var = sn * 0.03;
		vFragColor = vec4(
			0.90 + smooth_var,
			0.82 + smooth_var,
			0.82 + smooth_var,
			0.92);  // slightly translucent
		specMult = 0.55;
	}
	else if(material>10) {  // tendon (material 11)
		// Pearlescent white with fibrous longitudinal texture
		// Fibers run along the t (v) texture coordinate direction
		faceUV = vTexCoords;
		// Longitudinal fiber pattern: high frequency along t, low along s
		float fiber = snoise(vec2(faceUV.s * 60.0, faceUV.t * 8.0));
		float fiberFine = snoise(vec2(faceUV.s * 120.0, faceUV.t * 12.0));
		float fibrosity = fiber * 0.06 + fiberFine * 0.03;
		// Pearlescent base: bright white with slight iridescence
		float pearlBase = 0.94;
		vFragColor = vec4(
			pearlBase + fibrosity,
			pearlBase - 0.01 + fibrosity,
			pearlBase - 0.03 + fibrosity * 0.8,
			1.0);
		// Fiber-aligned normal perturbation for glistening effect
		float fiberNx = snoise(vec2((faceUV.s + fatIncr) * 60.0, faceUV.t * 8.0))
		              - snoise(vec2((faceUV.s - fatIncr) * 60.0, faceUV.t * 8.0));
		normDelta = normalize(vec3(fiberNx * 0.4, 0.0, 1.0));
		specMult = 0.75;
	}
	else if(material>9) {  // undermined skin or periosteum (material 10)
		vFragColor = vec4(0.3882, 0.6706, 0.9333, 1.0);
		specMult = 0.8;
	}
	else if(material>6) {  // periosteum (material 7-8, undermined 8, not undermined 7)
		// Shoulder periosteum: cream/white bone surface lining
		vFragColor = vec4(0.96, 0.93, 0.87, 1.0);
		specMult = 0.5;
	}
	else if(material>5) {  // deep cut muscle - deltoid (material 6)
		// Brighter red skeletal muscle with striated texture
		faceUV = vTexCoords * 80.0f;
		nei[0] = snoise(faceUV - fatD);
		nei[1] = snoise(faceUV + fatD);
		nei[2] = snoise(faceUV - fatD.yx);
		nei[3] = snoise(faceUV + fatD.yx);
		getFat(nei,vFragColor,normDelta,specMult);
		// Deltoid muscle: brighter red, more saturated than facial muscle
		// Add striation effect along one direction
		float striation = snoise(vec2(vTexCoords.s * 150.0, vTexCoords.t * 20.0)) * 0.06;
		vFragColor = vec4(
			vFragColor.r * 0.80 + striation,
			vFragColor.r * 0.18 + striation * 0.3,
			vFragColor.r * 0.18 + striation * 0.3,
			1.0);
	}
	else if(material>4) {  // deep tissue bed - fascial plane (material 5)
		// Shoulder fascia: white/silvery connective tissue
		faceUV = vTexCoords * 150.0;
		sn = snoise(faceUV);
		float fasciaSn = snoise(faceUV * 0.5);
		// White-silver base with subtle fibrous variation
		float fasciaBase = 0.88 + 0.04 * fasciaSn;
		vFragColor = vec4(
			fasciaBase + sn * 0.02,
			fasciaBase + sn * 0.02,
			fasciaBase + 0.02 + sn * 0.02,
			1.0);
		// Gentle normal perturbation for fibrous look
		float nx = snoise(faceUV + vec2(fatIncr, 0.0)) - snoise(faceUV - vec2(fatIncr, 0.0));
		float ny = snoise(faceUV + vec2(0.0, fatIncr)) - snoise(faceUV - vec2(0.0, fatIncr));
		normDelta = normalize(vec3(nx * 0.3, ny * 0.3, 1.0));
		specMult = 0.35;
        }
	else if(material>3) {  // flap bottom fat / subcutaneous (material 4)
		faceUV = vTexCoords*400.0f;
		nei[0] = snoise(faceUV - fatD);
		nei[1] = snoise(faceUV + fatD);
		nei[2] = snoise(faceUV - fatD.yx);
		nei[3] = snoise(faceUV + fatD.yx);
		getFat(nei,vFragColor,normDelta,specMult);
	}
	else if(material>2) {  // incised skin edge (material 3)
		faceUV = vTexCoords;
		sn = snoise(vec2(dm*faceUV.t*3.0,0.5));
		float val = (sn+1.0)*0.5;
		if(0.415 + 0.2*val < faceUV.s) {  // fat
			faceUV *= vec2(6.4,dm*4.7);
			nei[0] = snoise(faceUV - fatD);
			nei[1] = snoise(faceUV + fatD);
			nei[2] = snoise(faceUV - fatD.yx);
			nei[3] = snoise(faceUV + fatD.yx);
			getFat(nei,vFragColor,normDelta,specMult);
		}
		else if(0.40 + 0.2*val < faceUV.s) {  // dermal-fat junction
			vFragColor = vec4(0.51, 0.44, 0.1412, 1.0);
			specMult = 0.2; }
		else {  // dermis
			specMult = 0.35;
			if(faceUV.s<0.05)
				vFragColor = vec4(0.71, 0.57255, 0.2784, 1.0);
			else if(faceUV.s<0.07)
				vFragColor = vec4(0.843, 0.737, 0.51, 1.0);
			else {
				sn = snoise(vec2((dm*faceUV.t+0.4)*4.2,0.5));
				val = (sn-0.5)*2.0;
				if(0.30 + 0.05*val < faceUV.s) {
					vFragColor = vec4(0.9569, 0.8902, 0.71, 1.0); }
				else {
					vFragColor = vec4(0.7255, 0.5059, 0.2039, 1.0);
					vFragColor = vFragColor*(0.5+2.8*faceUV.s); }
			}
		}
        }
	else if(material>1) {	// top of skin (material 2)
		// Shoulder skin: slightly more tan, less pink than facial skin
		vec4 tx1 = texture(normalMap,vTexCoords.st);
		tx1.rgb -= vec3(0.5);
		normDelta = tx1.rgb*2.0;
		specMult = 0.11;
		vFragColor = texture(colorMap, vTexCoords.st);
		// Shift toward tan: reduce red slightly, boost green very slightly
		vFragColor.r *= 0.97;
		vFragColor.g *= 1.02;
	}
	else { // blank end material==1
		vec4 tx1 = vec4(1.0);
		tx1.rgb = vec3(0.1,0.18,0.33);
		normDelta = vec3(1.0);
		specMult = 0.0;
		vFragColor = tx1;
	}
	lightVal += diffuseVal*max(dot(normDelta,vLightDir), 0.0);
	if(material==1)
		vFragColor *= 0.8 + 0.2*lightVal;
	else
		vFragColor *= lightVal;
	vec3 reflectDir = reflect(vLightDir,normDelta);
	float spec = max(dot(vEyeDir,reflectDir),0.0);
	spec = pow(spec,40.0);
	spec *= specMult;
	litColor = min(vFragColor.rgb + spec, vec3(1.0));
	vFragColor = vec4(litColor, 1.0);
}
